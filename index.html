async function handleCopyAction() { const copyBtn = document.getElementById('copyBtn'); const originalBtnText = copyBtn.innerHTML; copyBtn.innerHTML = `⏳ Uploading Photos & Generating Links...`; copyBtn.disabled = true; const partner = document.getElementById('tradePartner').value || 'Trade Partner'; const addr = document.getElementById('address').value || 'Job Site'; const date = document.getElementById('woDate').value; const dueDate = document.getElementById('woDueDate').value; const woNum = document.getElementById('woNumber').value; const supervisor = document.getElementById('supervisor').value; savePartner(partner); let txt = `BIDDISON HOMES - WORK ORDER (${woNum})\n`; txt += `Date: ${date} | Address: ${addr}\n`; if (dueDate) { const [y, m, d] = dueDate.split('-'); txt += `COMPLETE BY: ${m}/${d}/${y}\n`; } txt += `Trade Partner: ${partner} | Supervisor: ${supervisor}\n\nTasks:\n`; const rows = document.getElementById('tasksContainer').children; Array.from(rows).forEach((row, idx) => { const desc = row.querySelector('.task-desc').value; const loc = row.querySelector('.task-loc').value; if (desc) txt += `${idx + 1}. ${desc} ${loc ? '(' + loc + ')' : ''}\n`; }); const notes = document.getElementById('notes').value; if (notes) txt += `\nSpecial Instructions:\n${notes}\n`; // 1. Wait for Apps Script to upload photos and return direct links const driveResult = await saveToGoogleDrive(woNum, addr, partner, dueDate, txt); if (driveResult && driveResult.photoLinks && driveResult.photoLinks.length > 0) { txt += `\nSite Photos:\n` + driveResult.photoLinks.join('\n') + `\n`; } // 2. iOS-Compatible Copy Handler let copySuccessful = false; // Try Clipboard API if (navigator.clipboard && navigator.clipboard.writeText) { try { await navigator.clipboard.writeText(txt); copySuccessful = true; } catch (err) { copySuccessful = false; } } // Fallback for iOS Safari when async write is blocked if (!copySuccessful) { const tempTextArea = document.createElement("textarea"); tempTextArea.value = txt; // Prevent keyboard pop-up on iOS while selecting tempTextArea.contentEditable = true; tempTextArea.readOnly = false; tempTextArea.style.position = "absolute"; tempTextArea.style.left = "-9999px"; document.body.appendChild(tempTextArea); // Selection range specific to iOS const range = document.createRange(); range.selectNodeContents(tempTextArea); const selection = window.getSelection(); selection.removeAllRanges(); selection.addRange(range); tempTextArea.setSelectionRange(0, 999999); try { copySuccessful = document.execCommand("copy"); } catch (e) { copySuccessful = false; } document.body.removeChild(tempTextArea); } copyBtn.innerHTML = originalBtnText; copyBtn.disabled = false; if (copySuccessful) { alert(`Work Order ${woNum} copied to clipboard! Ready to paste into SMS or Email.`); } else { // Ultimate fallback: Populate output area and auto-select for manual copy on iOS const outputContainer = document.getElementById('outputContainer'); const outputText = document.getElementById('outputText'); if (outputContainer && outputText) { outputContainer.classList.remove('hidden'); outputText.value = txt; outputText.focus(); outputText.select(); alert(`Work Order ${woNum} generated! Please tap "Copy" on the highlighted text box below.`); } else { alert(`Work Order ${woNum} generated, but automatic clipboard copy was blocked. Please check permissions.`); } } attachedPhotos = []; renderPhotoPreviews(); incrementWOCounter(); }
